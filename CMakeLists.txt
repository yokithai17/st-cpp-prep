cmake_minimum_required(VERSION 3.20)
project(st_cpp_prep LANGUAGES CXX)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(environment_check tools/environment_check.cpp)
target_compile_features(environment_check PRIVATE cxx_std_20)
set_target_properties(environment_check PROPERTIES CXX_EXTENSIONS OFF)

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(environment_check PRIVATE -Wall -Wextra -Wpedantic)
elseif(MSVC)
    target_compile_options(environment_check PRIVATE /W4 /Zc:__cplusplus)
endif()

include(CTest)
if(BUILD_TESTING)
    add_test(NAME environment_check COMMAND environment_check)

    add_executable(ring_buffer_tests exercises/03-ring-buffer/test.cpp)
    set_target_properties(ring_buffer_tests PROPERTIES CXX_EXTENSIONS OFF)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(ring_buffer_tests PRIVATE -Wall -Wextra -Wpedantic)
    elseif(MSVC)
        target_compile_options(ring_buffer_tests PRIVATE /W4 /Zc:__cplusplus)
    endif()
    add_test(NAME ring_buffer_tests COMMAND ring_buffer_tests)

    if(UNIX)
        add_executable(unique_fd_tests exercises/01-unique-fd/tests.cpp)
        target_compile_features(unique_fd_tests PRIVATE cxx_std_20)
        set_target_properties(unique_fd_tests PROPERTIES CXX_EXTENSIONS OFF)
        if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(unique_fd_tests PRIVATE -Wall -Wextra -Wpedantic)
        endif()
        add_test(NAME unique_fd_tests COMMAND unique_fd_tests)
    endif()
endif()
